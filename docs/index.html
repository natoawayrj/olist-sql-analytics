1. Coorte de Retenção e Intervalo de Recompra (sql/02_coorte_retencao_recompra.sql)

CTEs + ROW_NUMBER(), LAG() e FIRST_VALUE() sobre customer_unique_id.

WITH sequencia_pedidos AS (
    SELECT
        customer_unique_id,
        order_id,
        order_purchase_timestamp,
        ROW_NUMBER() OVER (
            PARTITION BY customer_unique_id
            ORDER BY order_purchase_timestamp, order_id
        ) AS ordem_compra,
        LAG(order_purchase_timestamp) OVER (
            PARTITION BY customer_unique_id
            ORDER BY order_purchase_timestamp, order_id
        ) AS compra_anterior
    FROM vw_pedidos_consolidados
    WHERE order_status = 'delivered'
),
retencao AS (
    SELECT
        safra_inicial,
        mes_indice,
        clientes_ativos,
        ROUND(
            clientes_ativos * 100.0 / FIRST_VALUE(clientes_ativos) OVER (
                PARTITION BY safra_inicial ORDER BY mes_indice
            ), 2
        ) AS pct_retencao
    FROM agregado_coorte
)

2. Curva de Pareto 80/20 (sql/03_pareto_categorias_sellers.sql)

DENSE_RANK() + soma acumulada em janela móvel (ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW).

WITH receita_categoria AS (
    SELECT categoria_raw, ROUND(SUM(receita_item), 2) AS receita_total
    FROM vw_itens_entregues
    GROUP BY categoria_raw
)
SELECT
    DENSE_RANK() OVER (ORDER BY receita_total DESC) AS ranking_receita,
    categoria_raw,
    receita_total,
    ROUND(
        SUM(receita_total) OVER (
            ORDER BY receita_total DESC
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) * 100.0 / SUM(receita_total) OVER (),
        2
    ) AS pct_receita_acumulada
FROM receita_categoria;

3. SLA Logístico & Ranking de UFs (sql/04_sla_entrega_vs_avaliacao.sql)

Proporção de volume com SUM(COUNT(*)) OVER () e ranqueamento com RANK() OVER.

SELECT
    faixa_sla,
    COUNT(*) AS qtd_pedidos,
    ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (), 2) AS pct_volume,
    ROUND(AVG(nota_media_pedido), 2) AS nota_media,
    ROUND(AVG(CASE WHEN eh_detrator = 1 THEN 100.0 ELSE 0.0 END), 1) AS pct_detratores
FROM vw_pedidos_consolidados
WHERE order_status = 'delivered'
  AND nota_media_pedido IS NOT NULL
GROUP BY ordem_faixa_sla, faixa_sla
ORDER BY ordem_faixa_sla;
